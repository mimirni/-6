#include <iostream>
using namespace std;

int main() {
    int userNumber;
    cout << "[ + ] Введите число: ";
    cin >> userNumber;
    
    for (int i = 1; i <= 9; i++) {
        cout << "[ + ] Решите пример: " << userNumber << " x " << i << endl;
        
        int userAnswer;
        cout << "[ + ] Ответ: ";
        cin >> userAnswer;
        
        int correctAnswer = userNumber * i;
        
        if (userAnswer == correctAnswer) {
            cout << "[ + ] Пример решен правильно!" << endl;
        } else {
            cout << "[ - ] Ошибка, пример решен неверно!" << endl;
            return 0; 
        }
    }
    
    cout << "[ + ] Примеры решены, поздравляем!" << endl;
    
    return 0;
}
